Depends on the property type. Using your existing `notionFetch`/`NotionEnv` pattern from [client.ts](curriculum-app/lib/notion/client.ts):

**Checkbox property** (e.g. `Checked`):
```ts
import { NotionEnv, notionFetch } from './client';

export async function countChecked(databaseId: string, propertyName = 'Checked'): Promise<number> {
  let count = 0;
  let cursor: string | undefined;

  do {
    const res = await notionFetch<{ results: any[]; has_more: boolean; next_cursor: string | null }>(
      `/databases/${databaseId}/query`,
      {
        method: 'POST',
        body: {
          page_size: 100,
          start_cursor: cursor,
          filter: { property: propertyName, checkbox: { equals: true } },
        },
        revalidate: 0,
      },
    );
    count += res.results.length;
    cursor = res.has_more ? res.next_cursor! : undefined;
  } while (cursor);

  return count;
}
```

**Status property** (e.g. Status = "Done", following the pattern in [tasks.ts:64-96](curriculum-app/lib/notion/tasks.ts:64)):
```ts
export async function countByStatus(databaseId: string, statusValue: string): Promise<number> {
  let count = 0;
  let cursor: string | undefined;

  do {
    const res = await notionFetch<{ results: any[]; has_more: boolean; next_cursor: string | null }>(
      `/databases/${databaseId}/query`,
      {
        method: 'POST',
        body: {
          page_size: 100,
          start_cursor: cursor,
          filter: { property: 'Status', status: { equals: statusValue } },
        },
        revalidate: 0,
      },
    );
    count += res.results.length;
    cursor = res.has_more ? res.next_cursor! : undefined;
  } while (cursor);

  return count;
}
```
