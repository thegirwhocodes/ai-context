## FULL exact content of `supabase-sabi-telephony.sql`

```sql
-- Sabi Telephony: call state, SMS log, phone-based students, audio storage
-- Idempotent migration. Run in Supabase Dashboard > SQL Editor.
--
-- Recreates the 3 DB objects (+ 1 storage bucket) that the voice code depends on
-- but that were never committed to a SQL file:
--   - table  sabi_active_calls   (lib/voice/call-state.ts)
--   - table  sabi_sms_log        (lib/voice/sms.ts)
--   - column sabi_students.phone_number  (lib/voice/sabi-shared.ts)
--   - bucket sabi-audio (PUBLIC)          (lib/voice/audio-storage.ts)
--
-- RLS note: like sabi_students / sabi_sessions / lesson_completions, these tables
-- are only ever touched server-side with the SUPABASE_SERVICE_ROLE_KEY, which
-- bypasses RLS. We DISABLE ROW LEVEL SECURITY explicitly (matching
-- supabase-lesson-completions.sql) so the service role works and no anon client
-- can reach the rows.

-- ---------------------------------------------------------------------------
-- 1. sabi_students.phone_number  (used by findOrCreateStudentByPhone)
-- ---------------------------------------------------------------------------
-- Phone is the production student identifier (browser_id was the demo identifier).
ALTER TABLE sabi_students
  ADD COLUMN IF NOT EXISTS phone_number TEXT;

-- findOrCreateStudentByPhone() inserts a student with ONLY phone_number set,
-- but the original schema declared browser_id TEXT UNIQUE NOT NULL. A phone-only
-- insert would violate that NOT NULL. Relax it so phone-based students are valid.
ALTER TABLE sabi_students
  ALTER COLUMN browser_id DROP NOT NULL;

-- Fast + unique lookup by phone. Partial unique index ignores NULLs so existing
-- browser_id-only (demo) students without a phone don't collide.
CREATE UNIQUE INDEX IF NOT EXISTS idx_sabi_students_phone_number
  ON sabi_students(phone_number)
  WHERE phone_number IS NOT NULL;

-- ---------------------------------------------------------------------------
-- 2. sabi_active_calls  (per-call conversation state; serverless-safe store)
-- ---------------------------------------------------------------------------
CREATE TABLE IF NOT EXISTS sabi_active_calls (
  call_sid     TEXT PRIMARY KEY,
  student_id   UUID REFERENCES sabi_students(id) ON DELETE SET NULL,
  phone_number TEXT,
  messages     JSONB NOT NULL DEFAULT '[]',
  started_at   TIMESTAMPTZ DEFAULT NOW(),
  updated_at   TIMESTAMPTZ DEFAULT NOW()
);

-- call_sid is the PK (already indexed). updated_at index speeds cleanupExpiredCalls().
CREATE INDEX IF NOT EXISTS idx_sabi_active_calls_updated_at
  ON sabi_active_calls(updated_at);

ALTER TABLE sabi_active_calls DISABLE ROW LEVEL SECURITY;

-- ---------------------------------------------------------------------------
-- 3. sabi_sms_log  (outbound/inbound SMS log + practice-answer tracking)
-- ---------------------------------------------------------------------------
CREATE TABLE IF NOT EXISTS sabi_sms_log (
  id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  student_id      UUID REFERENCES sabi_students(id) ON DELETE SET NULL,
  phone_number    TEXT NOT NULL,
  direction       TEXT NOT NULL DEFAULT 'outbound',  -- 'outbound' | 'inbound'
  message         TEXT,
  sms_type        TEXT DEFAULT 'general',            -- 'general'|'summary'|'practice'|'reminder'
  channel         TEXT,                              -- 'at' | 'twilio'
  expected_answer TEXT,                              -- set for practice questions
  actual_answer   TEXT,                              -- filled in when student replies
  is_correct      BOOLEAN,                           -- result of checkPracticeAnswer
  created_at      TIMESTAMPTZ DEFAULT NOW()
);

-- checkPracticeAnswer() filters by phone_number + sms_type + direction + actual_answer IS NULL,
-- ordered by created_at DESC. This composite index covers that hot path.
CREATE INDEX IF NOT EXISTS idx_sabi_sms_log_lookup
  ON sabi_sms_log(phone_number, sms_type, direction, created_at DESC);

CREATE INDEX IF NOT EXISTS idx_sabi_sms_log_student_id
  ON sabi_sms_log(student_id);

ALTER TABLE sabi_sms_log DISABLE ROW LEVEL SECURITY;

-- ---------------------------------------------------------------------------
-- 4. sabi-audio storage bucket  (PUBLIC; serves TTS audio to Africa's Talking <Play>)
-- ---------------------------------------------------------------------------
-- Create the bucket (idempotent). public = true so getPublicUrl() returns a URL
-- that AT can fetch without auth.
INSERT INTO storage.buckets (id, name, public)
VALUES ('sabi-audio', 'sabi-audio', true)
ON CONFLICT (id) DO UPDATE SET public = true;

-- Allow anyone to READ objects in this bucket (needed for AT to download the audio).
-- Uploads/deletes go through the service role, which bypasses these policies.
DROP POLICY IF EXISTS "sabi_audio_public_read" ON storage.objects;
CREATE POLICY "sabi_audio_public_read"
  ON storage.objects FOR SELECT
  USING (bucket_id = 'sabi-audio');

SELECT 'sabi telephony schema (calls, sms_log, phone_number, sabi-audio bucket) ready!' AS status;
```

## Bucket-creation steps (if the SQL in section 4 fails)

Section 4 creates the bucket via SQL, which works in most Supabase projects. If your project restricts `storage.buckets` / `storage.objects` writes from the SQL editor (some do), create it from the dashboard instead:

1. Supabase Dashboard → **Storage** → **New bucket**
2. Name: `sabi-audio`
3. Toggle **Public bucket** = ON
4. Create. No additional policies are required for uploads/deletes (the server uses `SUPABASE_SERVICE_ROLE_KEY`, which bypasses RLS); public read is granted automatically for a public bucket, so the `sabi_audio_public_read` policy in section 4 is only needed if you keep the bucket private.
