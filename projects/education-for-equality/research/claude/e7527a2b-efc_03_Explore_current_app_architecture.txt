```python
def gen_signing_feed():
    while True:
        with _signing_lock:
            jpg = _signing_jpeg
        if jpg is None:
            time.sleep(0.05)
            continue
        yield (b'--frame\r\nContent-Type: image/jpeg\r\n\r\n' + jpg + b'\r\n')
        time.sleep(0.033)  # ~30fps delivery rate

def gen_speaking_feed():
    while True:
        with _speaking_lock:
            jpg = _speaking_jpeg
        if jpg is None:
            time.sleep(0.05)
            continue
        yield (b'--frame\r\nContent-Type: image/jpeg\r\n\r\n' + jpg + b'\r\n')
        time.sleep(0.033)
```
